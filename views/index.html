    Регистрация и аутентификация¶

    Регистрация (register)¶

    Обрабатывает регистрацию новых пользователей.

    def register(request):
    if request.method == 'POST':
        form = UserRegistrationForm(request.POST)
        if form.is_valid():
            user = form.save(commit=False)
            user.set_password(form.cleaned_data['password'])
            user.save()
            return redirect('home')
    else:
        form = UserRegistrationForm()
    return render(request, 'register.html', {'form': form})

    Вход (custom_login)¶

    Обрабатывает аутентификацию пользователей.

    def custom_login(request):
    if request.method == 'POST':
        form = AuthenticationForm(data=request.POST)
        if form.is_valid():
            user = form.get_user()
            login(request, user)
            next_url = request.GET.get('next', '/')
            return redirect(next_url)
    else:
        form = AuthenticationForm()

    return render(request, 'login.html', {'form': form})

    Выход (custom_logout)¶

    Завершает сеанс пользователя

    def custom_logout(request):
    logout(request)
    return redirect('/')

    Гонки¶

    Домашняя страница (home)¶

    Отображает ближайшие гонки

    def home(request):
    races = Race.objects.filter(date__gte=datetime.date.today())
    return render(request, 'home.html', {
        'races': races,
    })

    На этой странице также выполнена пагинация, таблица с данными о предстоящих гонках

    Детали гонки (race_detail)¶

    Отображает подробности гонки, включая комментарии, результаты и регистрации.

    def race_detail(request, id):
    race = get_object_or_404(Race, id=id)
    comments = RaceComment.objects.filter(race_date=race.date)
    results = RaceResult.objects.filter(race=race)
    registrations = Registration.objects.filter(race=race)
    today = timezone.localdate()

    confirmed_registrations = registrations.filter(is_confirmed=True)
    unconfirmed_registrations = registrations.filter(is_confirmed=False)
    registration = Registration.objects.filter(race=race, user=request.user).first()
    is_registered = registration is not None

    return render(request, 'race_detail.html', {
        'race': race,
        'results': results,
        'confirmed_registrations': confirmed_registrations,
        'unconfirmed_registrations': unconfirmed_registrations,
        'is_registered': is_registered,
        'registration': registration,
        'comments': comments,
        'today': today
    })

    Список гонок (race_list)¶

    Предоставляет пагинированный список гонок с поиском.

    def race_list(request):
    search_query = request.GET.get('search', '')
    races = Race.objects.all()

    if search_query:
        races = races.filter(name__icontains=search_query)

    paginator = Paginator(races, 5)
    page_number = request.GET.get('page')
    page_obj = paginator.get_page(page_number)

    return render(request, 'race_list.html', {'page_obj': page_obj, 'search_query': search_query})

    Регистрация на гонку (register_for_race)¶

    Позволяет пользователю зарегистрироваться на гонку

    def register_for_race(request, race_id):
    race = Race.objects.get(id=race_id)
    user = request.user

    try:
        racer = Racer.objects.get(user=user)
    except Racer.DoesNotExist:
        return redirect('register')

    if Registration.objects.filter(user=user, race=race).exists():
        return redirect('already_registered')

    registration = Registration(user=user, race=race, racer=racer)
    registration.save()

    return redirect('race_detail', id=race.id)

    Добавление комментариев к гонке (add_race_comment)¶

    Позволяет пользователям оставлять комментарии к гонкам

    @login_required
def add_race_comment(request, race_id):
    race = get_object_or_404(Race, id=race_id)
    if request.method == 'POST':
        form = RaceCommentForm(request.POST)
        if form.is_valid():
            comment = form.save(commit=False)
            comment.registration = Registration.objects.filter(user=request.user, race=race).first()
            comment.race_date = race.date
            comment.save()
            return redirect('race_detail', id=race_id)
    else:
        form = RaceCommentForm()

    return render(request, 'add_comment.html', {'race': race, 'form': form})

    Удаление регистрации на гонку (delete_registration)¶

    Удаляет регистрацию пользователя на гонку

    def delete_registration(request, race_id):
    race = get_object_or_404(Race, id=race_id)
    registration = get_object_or_404(Registration, user=request.user, race=race)

    if request.method == 'POST':
        registration.delete()
        return redirect('race_list')

    return render(request, 'delete_registration.html', {'race': race, 'registration': registration})
